// ウェーブテキスト / Wave text
// パック: motion2 (モーショングラフィックスの演出 第2弾) · element 10 · カテゴリ: 文字・図形の演出
// 何が起きるか: 「WAVE TEXT」の文字が1文字ずつ、波のように順に上下する。
// 動きの仕様:
//   - 「WAVE TEXT」（70px・太字・字間4px）。4.7秒に3回の波（1波1.567秒）
//   - i 文字目の縦の位置 = −28·sin(φ − 0.62i)。φ = 2π × 3 × 時間 ÷ 4.7（隣の文字は0.62radずつ遅れる）
//   - 上がっている文字ほど縦に伸ばし（1 + 0.06·s）、色を白→黄に寄せる（s = sin の値。0〜1に制限して混ぜる）
//   - 各文字の下に影の楕円。文字が上がるほど小さくする
// 使いどころ: ロゴ・短い見出し・BGMに合わせたテロップに。楽しさやリズム感を出したいとき。
// タグ: wave-text, wave, text, letters, stagger, bounce, kinetic-type
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit motion2 -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 動く見本: https://rainman016.github.io/motion-prompt-zh/?lang=ja#130
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 10
// 10 ウェーブテキスト（1文字ずつ、波のように順に上下する）
CELLS[9] = (k, w, h, u) => {
  stage(k);
  const f = `800 70px ${DISP}`, cs = chars(k, 'WAVE TEXT', f, 4), Y = h / 2 + 2, ph = 2 * Math.PI * 3 * u / CYC;
  cs.forEach((c, i) => {
    if (c.ch === ' ') return;
    const s = Math.sin(ph - i * 0.62), y = -28 * s;
    k.fillStyle = 'rgba(0,0,0,.3)'; k.beginPath(); k.ellipse(w / 2 + c.x, Y + 62, c.w * (0.42 - 0.12 * s), 4, 0, 0, 7); k.fill();
    k.save(); k.translate(w / 2 + c.x, Y + y); k.scale(1, 1 + 0.06 * s); txt(k, c.ch, 0, 0, f, mix('#FFFFFF', C.yel, clamp(s)), 'center', 'middle'); k.restore();
  });
};
