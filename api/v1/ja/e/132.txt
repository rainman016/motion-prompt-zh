// 可変フォントアニメーション / Variable font animation
// パック: motion2 (モーショングラフィックスの演出 第2弾) · element 12 · カテゴリ: 文字・図形の演出
// 何が起きるか: 同じ単語の太さが、細→極太→細となめらかに変わり、下のつまみが太さを示す。
// 動きの仕様:
//   - 「MOTION」（112px・system-ui・字間2px）。太さを4.7秒に2往復: v = 0.5 − 0.5·cos(2π × 2 × 時間 ÷ 4.7)（cosなので両端でなめらかに止まる）
//   - 太さ = round(100 + 800v)（100〜900）をそのままフォントの太さに入れる
//   - 下にスライダー（x=112→427・y=180）。左に「細」、右に「太」。つまみ（黄・半径14）が v に合わせて動き、上に今の太さの数字を出す
// 使いどころ: タイトル・ロゴ・強調したい一語に。文字の形そのものを動かして、音や感情の強弱を見せたいとき。
// タグ: variable-font, font-weight, typography, text, weight, kinetic-type
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit motion2 -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 動く見本: https://rainman016.github.io/motion-prompt-zh/?lang=ja#132
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 12
// 12 可変フォントアニメーション（同じ単語の太さが、細→極太→細となめらかに変わる）
CELLS[11] = (k, w, h, u) => {
  stage(k);
  const v = 0.5 - 0.5 * Math.cos(2 * Math.PI * 2 * u / CYC), wt = Math.round(100 + 800 * v);
  txt(k, 'MOTION', w / 2, 80, `${wt} 112px system-ui`, '#fff', 'center', 'middle', '2px');
  const x0 = 112, x1 = w - 112, y = 180, kx = x0 + (x1 - x0) * v;
  txt(k, '細', x0 - 34, y + 1, `300 30px ${JP}`, C.dim, 'center', 'middle');
  txt(k, '太', x1 + 34, y + 1, `900 30px ${JP}`, C.dim, 'center', 'middle');
  k.fillStyle = C.navy3; rr(k, x0, y - 5, x1 - x0, 10, 5); k.fill();
  k.fillStyle = C.blue; rr(k, x0, y - 5, Math.max(10, kx - x0), 10, 5); k.fill();
  k.fillStyle = C.yel; k.beginPath(); k.arc(kx, y, 14, 0, 7); k.fill();
  txt(k, String(wt), kx, y - 30, `800 24px ${DISP}`, C.yel, 'center', 'middle');
};
