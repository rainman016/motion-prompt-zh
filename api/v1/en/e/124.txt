// Wiggle / ウィグル
// Pack: motion2 (Motion graphics moves, part 2) · element 04 · Category: Motion techniques
// What happens: A bell keeps jittering irregularly: not one hit, but a small shake that never stops.
// Motion spec:
//   - A yellow bell pivoting at its top (centre, y = 34). It never stops moving.
//   - Angle = the sum of four sine waves: 23, 31, 43 and 59 cycles per 4.7 s (about 4.9, 6.6, 9.1 and 12.6 Hz), amplitudes 0.085, 0.065, 0.05 and 0.035 rad (up to about 0.235 rad, 13.5°, when they line up).
//   - Position also jitters: x with 29, 47 and 61 cycles (4, 3, 2 px), y with 37 and 53 cycles (2.4, 1.8 px).
//   - Every frequency is a whole number of cycles per 4.7 s, so the loop joins up; the counts share no common factor, so it does not look regular.
//   - The clapper (coral, radius 13) swings the other way (angle × 70 px). White vibration arcs (radius 86 and 110, width 6) flicker on both sides the whole time.
// When to use it: Use it on a notification, an incoming call or anything that should catch the eye, or to add a restless, handheld feel.
// Tags: wiggle, jitter, shake, random, noise, bell, notification
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://rainman016.github.io/motion-prompt-zh/?lang=en#124
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 04
// 04 ウィグル（ベルが、小刻みに不規則に揺れ続ける。1回の衝撃ではなく、ずっと続く細かい揺れ）
const WGF = 2 * Math.PI / CYC;
const wsum = (u, arr) => arr.reduce((s, [kk, am, ph]) => s + am * Math.sin(WGF * kk * u + ph), 0);
CELLS[3] = (k, w, h, u) => {
  stage(k);
  const rot = wsum(u, [[23, 0.085, 0.3], [31, 0.065, 1.9], [43, 0.05, 4.2], [59, 0.035, 2.6]]);
  const dx = wsum(u, [[29, 4, 0.7], [47, 3, 3.1], [61, 2, 5.0]]), dy = wsum(u, [[37, 2.4, 2.2], [53, 1.8, 0.4]]);
  const X = w / 2 + dx, Y = 34 + dy;
  // 両わきの振動の線（揺れているあいだ、ずっと出ている）
  k.strokeStyle = '#fff'; k.lineCap = 'round'; k.lineWidth = 6;
  [[86, 0.85], [110, 0.5]].forEach(([r, al], j) => { const fl = 0.55 + 0.45 * Math.abs(Math.sin(WGF * (41 + j * 12) * u)); k.globalAlpha = al * fl;[0, Math.PI].forEach(base => { k.beginPath(); k.arc(w / 2, 104, r, base - 0.42, base + 0.42); k.stroke(); }); });
  k.globalAlpha = 1;
  k.save(); k.translate(X, Y); k.rotate(rot);
  // 舌（ベルと逆にふれる）
  k.fillStyle = C.coral; k.beginPath(); k.arc(-rot * 70, 128, 13, 0, 7); k.fill();
  k.fillStyle = C.yel;
  k.beginPath(); k.arc(0, 8, 10, 0, 7); k.fill();
  k.beginPath(); k.moveTo(-50, 112); k.quadraticCurveTo(-42, 100, -42, 76); k.bezierCurveTo(-42, 36, -22, 18, 0, 18); k.bezierCurveTo(22, 18, 42, 36, 42, 76); k.quadraticCurveTo(42, 100, 50, 112); k.closePath(); k.fill();
  rr(k, -60, 106, 120, 16, 8); k.fill();
  k.fillStyle = 'rgba(255,255,255,.35)'; rr(k, -28, 40, 12, 50, 6); k.fill();
  k.restore();
};
