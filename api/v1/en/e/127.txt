// Ping-pong loop / ピンポンループ
// Pack: motion2 (Motion graphics moves, part 2) · element 07 · Category: Motion techniques
// What happens: A square travels to one end, then comes back along the same path as if played in reverse, again and again.
// Motion spec:
//   - One round trip per 4.7 s: 2.35 s forward, then the same motion played backwards for 2.35 s (progress g goes 0→1→0, linear).
//   - A mint rounded square (48 px, radius 13) hops from x = 92 to 447 along three arches of height 62·|sin(3πg)|, turning one full turn with g.
//   - Four ghost copies, 0.045 s apart, opacity 0.1–0.4.
//   - A playback bar below: blue with a play label while going forward, coral with a reverse label while going back. The progress bar and its knob follow g.
//   - Vertical posts mark the two turning points.
// When to use it: Use it for loaders, idle animations and any move that should repeat back and forth without a cut.
// Tags: ping-pong, loop, boomerang, yoyo, reverse, back-and-forth, loader
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://rainman016.github.io/motion-prompt-zh/?lang=en#127
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 07
// 07 ピンポンループ（端まで行ったら、同じ道を逆再生のように戻ってくる。往復をくり返す）
CELLS[6] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const q = u / CYC, fwd = q < 0.5, g = fwd ? 2 * q : 2 - 2 * q, X0 = 92, X1 = w - 92, base = 136, S = 48;
  const pos = gg => [lerp(X0, X1, gg), base - 62 * Math.abs(Math.sin(3 * Math.PI * gg))];
  k.save(); k.setLineDash([2, 12]); k.lineCap = 'round'; k.strokeStyle = C.navy3; k.lineWidth = 5; k.beginPath();
  for (let i = 0; i <= 150; i++) { const [px, py] = pos(i / 150); i ? k.lineTo(px, py - S / 2) : k.moveTo(px, py - S / 2); } k.stroke(); k.restore();
  k.fillStyle = C.navy3; rr(k, X0 - 40, base + 2, X1 - X0 + 80, 5, 2.5); k.fill();
  // 両端の折り返し点
  [X0, X1].forEach(xx => { k.fillStyle = C.navy3; rr(k, xx - 4, base - 70, 8, 72, 4); k.fill(); });
  // 残像
  for (let j = 4; j >= 1; j--) { const qq = mod(u - j * 0.045, CYC) / CYC, gg = qq < 0.5 ? 2 * qq : 2 - 2 * qq, [gx, gy] = pos(gg); k.globalAlpha = 0.1 * (5 - j); k.fillStyle = C.mint; k.save(); k.translate(gx, gy - S / 2); k.rotate(gg * Math.PI * 2); rr(k, -S / 2, -S / 2, S, S, 13); k.fill(); k.restore(); }
  k.globalAlpha = 1;
  const [x, y] = pos(g);
  k.save(); k.translate(x, y - S / 2); k.rotate(g * Math.PI * 2);
  k.fillStyle = C.mint; rr(k, -S / 2, -S / 2, S, S, 13); k.fill();
  k.fillStyle = '#fff'; k.beginPath(); k.arc(S / 2 - 13, -S / 2 + 13, 6.5, 0, 7); k.fill();
  k.restore();
  // 再生バー
  const ty = 184, tx0 = 196, tx1 = w - 34;
  k.fillStyle = fwd ? C.blue : C.coral; rr(k, 22, ty - 20, 158, 40, 20); k.fill();
  txt(k, fwd ? '▶ 再生' : '◀ 逆再生', 101, ty + 1, `800 25px ${JP}`, '#fff', 'center', 'middle');
  k.fillStyle = C.navy3; rr(k, tx0, ty - 5, tx1 - tx0, 10, 5); k.fill();
  k.fillStyle = fwd ? C.blue : C.coral; rr(k, tx0, ty - 5, Math.max(10, (tx1 - tx0) * g), 10, 5); k.fill();
  k.fillStyle = C.yel; k.beginPath(); k.arc(tx0 + (tx1 - tx0) * g, ty, 12, 0, 7); k.fill();
};
