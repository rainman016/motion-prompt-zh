// Text on a path / テキストオンパス
// Pack: motion2 (Motion graphics moves, part 2) · element 11 · Category: Text & shapes
// What happens: A line of text flows along a wavy line, each letter turning to follow it.
// Motion spec:
//   - Path: y = 118 + 42·sin(2πx ÷ 330 + 0.6) for x = −70 to 609, shown 6 px below as a navy dashed line (14/10, width 5).
//   - "ALONG THE LINE • " (42 px, bold, 2 px letter spacing) is repeated. Each character is placed at its arc-length position on the path and rotated to the path direction.
//   - The text moves along the path by the length of one copy of the string every 4.7 s, so the loop joins up.
//   - Letters are white; the bullet is yellow.
// When to use it: Use it to run text along a shape: around a logo, along a road, river or wave, or as a looping band of text.
// Tags: text-on-path, path-text, curved-text, text, marquee, wave, scroll
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion2 -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// Live demo: https://rainman016.github.io/motion-prompt-zh/?lang=en#131
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 11
// 11 テキストオンパス（文字列が、波打つ線に沿って流れていく）
const TOP = (() => { const pts = []; for (let x = -70; x <= CW + 70; x += 4) pts.push([x, 118 + 42 * Math.sin(x * 2 * Math.PI / 330 + 0.6)]); return { pts, cum: pathLen(pts) }; })();
CELLS[10] = (k, w, h, u) => {
  stage(k);
  k.save(); k.setLineDash([14, 10]); k.strokeStyle = C.navy3; k.lineWidth = 5; k.lineCap = 'round'; k.beginPath(); TOP.pts.forEach((p, i) => i ? k.lineTo(p[0], p[1] + 6) : k.moveTo(p[0], p[1] + 6)); k.stroke(); k.restore();
  const f = `800 42px ${DISP}`, S = 'ALONG THE LINE • '; k.font = f;
  const ws = [...S].map(ch => k.measureText(ch).width + 2), run = ws.reduce((a, b) => a + b, 0), L = TOP.cum[TOP.cum.length - 1], shift = (u / CYC) * run;
  for (let r = -2; r * run + shift < L; r++) {
    let s = r * run + shift;
    [...S].forEach((ch, j) => {
      const c = s + ws[j] / 2; s += ws[j];
      if (c < 0 || c > L || ch === ' ') return;
      const p = ptAt(TOP.pts, TOP.cum, c / L);
      k.save(); k.translate(p[0], p[1]); k.rotate(p[2]); txt(k, ch, 0, 0, f, ch === '•' ? C.yel : '#fff', 'center', 'alphabetic'); k.restore();
    });
  }
};
