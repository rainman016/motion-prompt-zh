// 横向滚动 / Horizontal scroll
// 分类包: lp (落地页与网页动效) · element 05 · 子分类: 滚动效果
// 动画表现: 页面向下滚动时分区固定，四张带编号的作品卡片向左移动。
// 运动规格:
//   - 滚动：0.3–3.0 秒为 0→420 px，3.5–4.5 秒为 420→0，采用三次缓入缓出。
//   - “WORKS”标题滚走，滚动到 40 px 时分区到达顶部并固定。
//   - 四张卡片各 200 × 112，间隔 20，两侧留白 34；横移距离 = 34·2 + 200·4 + 20·3 − 511 = 417 px。
//   - 卡片 x = 初始 x − p × 417，p = (scroll − 40) / 380；与滚动线性对应，无缓动。
//   - 卡片横移时滚动条滑块仍向下移动。
// 适用场景: 适合在纵向长页中成排展示作品、案例或产品。
// 标签: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#077
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
