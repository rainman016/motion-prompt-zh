// 遮罩转场 / Mask transition
// 分类包: motion (动态图形基础) · element 17 · 子分类: 转场与强调
// 动画表现: 圆形或星形窗口不断扩大，显露其内部的下一场景。
// 运动规格:
//   - 日景→夜景→日景，各占 2.35 秒；每半段的 0.45–1.6 秒执行转场，持续 1.15 秒，采用三次缓入缓出。
//   - 下一场景绘制在当前场景上方，并裁剪到形状内部；形状不断放大直至覆盖画面。
//   - 日转夜：以太阳中心 (112, 104) 为圆心，圆半径 0→470。
//   - 夜转日：以月亮中心 (427, 104) 为中心，五角星外半径 0→1150，同时旋转 1.2 rad，约 69°。
//   - 扩张时形状边缘带线宽 7 的白色轮廓。
// 适用场景: 适合从画面中的太阳、标志或按钮等对象发起场景切换。
// 标签: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#017
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
