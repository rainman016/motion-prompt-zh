// 焦点转移 / Rack focus
// 分类包: camera (镜头运动与剪辑) · element 09 · 子分类: 拍摄风格与时间
// 动画表现: 相机静止，焦点由前景花朵转向后方人物，二者的虚实关系互换。
// 运动规格:
//   - 位置、朝向、焦距均固定，机位高 1.1 m；只改变模糊程度。
//   - 前景花朵在左侧 0.75 m、距离 2 m，模糊 0 px → 7 px；人物与背景从 6 px → 0 px，人物在右侧 0.9 m、距离 9 m；地面网格的模糊量为背景的 0.6×。
//   - 1.0 秒至 2.6 秒变焦点，持续 1.6 秒，采用三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
//   - 蓝色角括号焦点框沿直线从花移向人物，半宽由 46 px → 30 px。
//   - 花朵和人物都不移动。
//   - 2.6–4.2 秒停留，最后 0.5 秒交叉淡化复位；右上角显示对焦距离 ◎ 2m → ◎ 9m。
// 适用场景: 适合不切镜地把注意力从前景引向背景，或反向引导。
// 标签: rack focus, focus pull, pull focus, depth of field, blur, bokeh, foreground
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/camera.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#033
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 09
// 09 ラックフォーカス
CELLS[8] = shot(camCell((k, w, h, p) => {
  world(k, w, h, cam({ y: 1.1 }), {
    person: { x: 0.9, z: 9 }, blurFg: 7 * p, blurBg: 6 * (1 - p),
    extra: [[-0.75, 0, 2.0, (kk, s) => { kk.fillStyle = C.g3; kk.fillRect(-0.025, 0, 0.05, 0.75); kk.fillStyle = C.blueM; for (let i = 0; i < 6; i++) { kk.beginPath(); kk.ellipse(Math.cos(i) * 0.09, 0.8 + Math.sin(i) * 0.09, 0.08, 0.05, i, 0, 7); kk.fill(); } kk.fillStyle = C.blue; kk.beginPath(); kk.arc(0, 0.8, 0.05, 0, 7); kk.fill(); }, true]],
    after: (kk, cc) => {   // ピント位置の枠
      const a = proj(cc, -0.75, 0.8, 2.0), b = proj(cc, 0.9, 0.9, 9);
      const x = lerp(a.x, b.x, p), y = lerp(a.y, b.y, p), r = lerp(46, 30, p);
      kk.strokeStyle = C.blue; kk.lineWidth = 2; kk.beginPath();
      [[-1, -1], [1, -1], [-1, 1], [1, 1]].forEach(([sx, sy]) => { kk.moveTo(x + sx * r, y + sy * r - sy * 9); kk.lineTo(x + sx * r, y + sy * r); kk.lineTo(x + sx * r - sx * 9, y + sy * r); });
      kk.stroke();
    }
  });
}, p => `◎ ${lerp(2, 9, p).toFixed(0)}m`, 1.0, 2.6));
