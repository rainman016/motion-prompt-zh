// 伦勃朗光 / Rembrandt lighting
// 分类包: photo (摄影光线与风格) · element 05 · 子分类: 布光与时段
// 动画表现: 侧面 45°、上方 45° 的灯亮起，右半脸进入阴影，暗侧脸颊只留一个倒三角亮区。
// 运动规格:
//   - 场景 1：地平线 y 150 前的头肩人物，头部中心 x 270、y 78、半径 29 px，不绘制五官和头发。 背景变暗，天空为 #E7E5E1。
//   - 头部 x = 270 + 0.04·29 右侧为阴影 #33405A，左侧为亮部 #FFC75A。
//   - 亮三角顶点为 (x 270+0.24R, y 78−0.12R)、(x 270+0.7R, y 78−0.12R)、(x 270+0.24R, y 78+0.5R)，R = 29，约宽 13 px、高 18 px；与主亮面间隔约宽 6 px 的阴影带。
//   - 身体中心左侧 −0.45×56 px 以外为亮部，右侧 +0.3×56 px 以外为暗部；灯位图为 45°，右上显示“45° · ↑45°”。
//   - 效果在 0.4 秒至 1.4 秒进入，保持到 4.0 秒，在 4.8 秒前消失；每轮 5 秒，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合具有绘画、雕塑和古典质感的人像。
// 标签: rembrandt lighting, portrait lighting, triangle, 45 degree light, chiaroscuro, painterly
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#101
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 05
// レンブラントライティング: 斜め45°・上からの光。影側の頬に三角形の光が残る
CELLS[4] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    kk.fillStyle = sh; kk.fillRect(PX + 0.04 * R, PY - R - 2, R * 2, R * 2 + 4);       // 鼻筋から右は影
    kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.clip();
    kk.fillStyle = lit; kk.beginPath();                                                   // 影側の頬に残る光の三角形
    kk.moveTo(PX + 0.24 * R, PY - 0.12 * R); kk.lineTo(PX + 0.7 * R, PY - 0.12 * R); kk.lineTo(PX + 0.24 * R, PY + 0.5 * R); kk.closePath(); kk.fill();
  });
  bust(k, PX, PY, R, 'rgba(0,0,0,0)', kk => {
    kk.fillStyle = 'rgba(0,0,0,0)';
    torso(kk, PX, PY, R, PX - W, PX - 0.45 * W, lit);
    torso(kk, PX, PY, R, PX + 0.3 * W, PX + W, sh);
  });
  plan(k, [[Math.PI / 4, a]]);
  hud(k, '45° · ↑45°');
});
