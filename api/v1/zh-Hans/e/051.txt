// 底部弹出面板 / Bottom sheet
// 分类包: ui (UI 组件) · element 03 · 子分类: 弹出与覆盖
// 动画表现: 手指轻点“查看详情”，面板从底部升起；向下拖动把手即可关闭。
// 运动规格:
//   - 542×191 卡片，周期 4.5 秒。手指用半径 17、白边的半透明圆表示，于 1.0 秒点击底部居中的 170×36“查看详情”胶囊按钮。
//   - 展开量在 1.05–1.45 秒由 0→1，使用 easeOutCubic；全卡片遮罩 rgba(18,20,28)，透明度 0.36·a。
//   - 白色面板高 136，上角半径 18，顶边 y = 191 − 136·a；阴影模糊 22、y 偏移 −4；顶部中央把手为 52×6。
//   - 内部有粗体标题和两行内容，配 26×26 图标。
//   - 关闭：手指于 2.75 秒落在把手 (271, 68)，在 3.0–3.6 秒下拖；a 乘以3.0–3.6 秒内的 (1 − easeInOutCubic)，面板随手指滑出底部。
// 适用场景: 适合移动端在保留当前页面背景时展示详情或选项。
// 标签: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#051
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('詳細を見る', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
