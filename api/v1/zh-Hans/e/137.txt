// 推移转场 / Push transition
// 分类包: motion2 (动态图形进阶) · element 17 · 子分类: 转场与强调
// 动画表现: 下一场景从侧面滑入，与当前场景一起运动并将其推出。
// 运动规格:
//   - 海景→山景→海景，各占 2.35 秒；每半段的 0.45–1.6 秒转场，持续 1.15 秒，采用三次缓入缓出。
//   - 两场景保持相接，同时移动整个画面宽度 539 px，无重叠、无淡化。
//   - 海转山向左移动，下一场景从右侧进入；山转海向右移动。
//   - 两场景接缝用线宽 6 的深蓝线标记。
// 适用场景: 适合连续前进的幻灯片、照片或步骤切换。
// 标签: push, push-transition, slide, transition, scene-change, carousel
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion2 -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion2.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#137
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 17
// 17 プッシュ（次の画面が横から入ってきて、前の画面を一緒に押し出す）
const SC17 = [scSea, scMtn];
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1)), dir = a === 0 ? -1 : 1;
  if (p <= 0) return SC17[a](k, w, h);
  if (p >= 1) return SC17[b](k, w, h);
  k.save(); k.translate(dir * p * w, 0); SC17[a](k, w, h); k.restore();
  k.save(); k.translate(dir * (p - 1) * w, 0); SC17[b](k, w, h); k.restore();
  const ex = dir < 0 ? w * (1 - p) : w * p;
  k.fillStyle = C.navy; k.fillRect(ex - 3, 0, 6, h);
});
