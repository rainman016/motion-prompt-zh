// 定格 / Freeze frame
// 分类包: camera (镜头运动与剪辑) · element 16 · 子分类: 拍摄风格与时间
// 动画表现: 人物跳至最高点时画面定格，1.4 秒后从同一帧继续。
// 运动规格:
//   - 机位完全固定，高 1.3 m；暂停的是素材时间。
//   - 1.6 秒定格，3.0 秒恢复，持续 1.4 秒；素材时间在 0–1.6 秒正常前进，定格时保持 1.6 秒，之后按 u-1.4 继续。
//   - 人物距离 7 m，素材时间 0.6–2.6 秒内按抛物线跳跃，峰值 1.1 m；定格恰好位于最高点，即进度 50%。
//   - 九片落叶同步暂停，其下落速度是 13 慢动作的 1.5×。
//   - 定格标记包括 5 px 蓝色边框和底部中央的暂停图标；图标为 68×40 px 白色圆角框内的两条蓝条，淡入淡出各 0.12 秒。
//   - 仅定格时右上角显示“FREEZE”；无交叉淡化复位。
// 适用场景: 适合强调决定性瞬间，或暂停以叠加名字和字幕。
// 标签: freeze frame, still frame, pause, hold, stop motion, highlight
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/camera.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#040
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 16
// 16 フリーズフレーム
CELLS[15] = shot((k, w, h, u) => {
  // 跳躍の頂点で 1.4 秒止める
  const tj = u < 1.6 ? u : u < 3.0 ? 1.6 : u - 1.4;   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6);
  const frozen = u >= 1.6 && u < 3.0;
  world(k, w, h, cam({ y: 1.3 }), { person: { z: 7, y: jumpY(q), armUp: q > 0 && q < 1 }, extra: leaves(tj, 1.5) });
  if (frozen) {
    const a = Math.min(1, (u - 1.6) / 0.12, (3.0 - u) / 0.12);
    k.save(); k.globalAlpha = a; k.strokeStyle = C.blue; k.lineWidth = 5; k.strokeRect(2.5, 2.5, w - 5, h - 5);
    k.fillStyle = 'rgba(255,255,255,.88)'; rr(k, w / 2 - 34, h - 58, 68, 40, 8); k.fill();
    k.fillStyle = C.blue; k.fillRect(w / 2 - 12, h - 49, 8, 22); k.fillRect(w / 2 + 4, h - 49, 8, 22); k.restore();
  }
  return frozen ? 'FREEZE' : '';
});
