// 懒加载 / Lazy loading
// 分类包: lp (落地页与网页动效) · element 18 · 子分类: 加载、导航与循环动效
// 动画表现: 灰色占位滚入后中央出现旋转指示器，图片到达后淡入；首行从一开始就已加载。
// 运动规格:
//   - 3 × 3 网格，每格 151 × 92，首行初始即显示；滚动在 0.35–1.1 秒为 0→110 px，1.7–2.5 秒为 110→225 px，3.5–4.5 秒为 225→0，均用三次缓入缓出。
//   - 占位顶边越过视口底部 168 px 时开始加载，每列比左侧列晚 0.12 秒。
//   - 占位中央的半径 11 旋转环显示 0.45 秒，每 0.8 秒转一圈。
//   - 随后图片透明度在 0.35 秒内从 0→1，采用三次缓出；未加载的占位显示图片图标。
//   - 占位离开视口底部时恢复未加载状态。
// 适用场景: 适合图片较多的页面，加快首屏显示。
// 标签: lazy-loading, lazy-load, images, placeholder, performance, spinner, gallery
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#090
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 18
// 18 遅延読み込み（画面に入った画像だけ、その時点で読み込まれて表示される）
CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'ギャラリー', 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {   // まだの画像: 灰色の枠と画像の印
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
