// 胶片颗粒 / Film grain
// 分类包: photo (摄影光线与风格) · element 22 · 子分类: 色调与后期风格
// 动画表现: 同一风景全图出现细颗粒，颜色略偏暖奶油色。
// 运动规格:
//   - 场景 3：左侧半身像的头部中心 x 140、y 82、半径 28 px；右侧有山丘、房屋、树与太阳，太阳 x 432、y 54、半径 15 px。
//   - 颗粒为 2,600 个固定随机种子的方块，尺寸 0.8–1.7 px；55% 为深色 #3A3530，45% 为白色。
//   - 颗粒透明度 16–38% × 效果强度；颗粒不移动、不闪烁。
//   - 所有颜色向 #F2DDBE 混合 16%。
//   - 效果在 0.4 秒至 1.4 秒进入，保持到 4.0 秒，在 4.8 秒前消失；每轮 5 秒，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合减弱纯净数码感，增加模拟胶片纹理。
// 标签: film grain, grain, noise, analog, 35mm film, texture, warm tone
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#118
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 22
// フィルムグレイン: 細かい粒子が全体に乗り、全体が少し暖かい色に寄る
CELLS[21] = cell((k, u) => {
  const a = env(u);
  const FILM = mapPal(BASE, c => mixc(c, '#F2DDBE', 0.16));
  const p = pal(BASE, FILM, a);
  const G = once('grain', () => { const r = rng(22), o = []; for (let i = 0; i < 2600; i++) o.push([r() * CW, r() * CH, 0.8 + r() * 0.9, r() < 0.55 ? 0 : 1, 0.16 + r() * 0.22]); return o; });
  const grain = kk => { kk.save(); for (const [x, y, s, d, al] of G) { kk.globalAlpha = al * a; kk.fillStyle = d ? '#FFFFFF' : '#3A3530'; kk.fillRect(x, y, s, s); } kk.restore(); };
  stageLand(k, p); grain(k);
});
