// 光标跟随 / Cursor follower
// 分类包: lp (落地页与网页动效) · element 11 · 子分类: 悬停与点击交互
// 动画表现: 光标在页面中移动，橙色圆稍微滞后地跟随。
// 运动规格:
//   - 光标轨迹 x = 255 + 175·sin(2πt/4.8)，y = 86 + 46·sin(4πt/4.8)，为横向八字，每 4.8 秒一圈。
//   - 跟随圆半径 17、橙色、透明度 0.88，绘制在光标下方。
//   - 每帧按 60 fps 计算，走完剩余距离的 12%，即 x += (cursor x − x) × 0.12。
//   - 箭头光标始终可见。
// 适用场景: 适合品牌站和作品集，增加互动与趣味。
// 标签: cursor, custom-cursor, cursor-follower, mouse-follow, lerp, trail, interaction
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#083
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 11
// 11 マウスストーカー（カーソルのあとを、丸が少し遅れて追いかける）
CELLS[10] = (k, w, h, u) => {
  frame(k);
  const cur = t => [255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC)];
  // 毎フレーム（60fps）残りの距離の 12% ずつ近づく
  let t = u - 2, [fx, fy] = cur(t);
  for (let i = 0; i < 120; i++) { t += 1 / 60; const [tx, ty] = cur(t); fx += (tx - fx) * 0.12; fy += (ty - fy) * 0.12; }
  view(k, () => {
    txt(k, 'CREATIVE STUDIO', VW / 2, 74, `800 34px ${EN}`, C.ink, 'center', 'alphabetic', '3px');
    bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, 0.7);
    btn(k, VW / 2 - 50, 128, 100, 24, '作品を見る', C.ink);
    k.fillStyle = C.acc; k.globalAlpha = 0.88; k.beginPath(); k.arc(fx, fy, 17, 0, 7); k.fill(); k.globalAlpha = 1;
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
