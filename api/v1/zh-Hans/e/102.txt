// 蝴蝶光 / Butterfly lighting
// 分类包: photo (摄影光线与风格) · element 06 · 子分类: 布光与时段
// 动画表现: 前方高位灯亮起，面部左右均匀受光，演示仅在下巴下方显示小而对称的阴影。
// 运动规格:
//   - 场景 1：地平线 y 150 前的头肩人物，头部中心 x 270、y 78、半径 29 px，不绘制五官和头发。 背景变暗，天空为 #E7E5E1。
//   - 头部完全填充亮色 #FFC75A，无侧面阴影。
//   - 下巴下方阴影为椭圆，中心 y 78+1.06R，宽 0.72R、高 0.38R；R = 29，约 21×11 px，颜色 #33405A。
//   - 身体中心 ±0.8×56 px 内为亮部，外侧细带保留底色；灯位图在相机一侧 0°，右上显示“0° · ↑45°”。
//   - 效果在 0.4 秒至 1.4 秒进入，保持到 4.0 秒，在 4.8 秒前消失；每轮 5 秒，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合美妆和时尚人像，修饰面部轮廓。
// 标签: butterfly lighting, paramount lighting, beauty lighting, portrait lighting, overhead light, glamour
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#102
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 06
// バタフライライティング: 顔の正面・高い位置からの光。顔は左右対称に明るく、あごの下にだけ小さな影
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();    // あごの下の影（左右対称）
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
