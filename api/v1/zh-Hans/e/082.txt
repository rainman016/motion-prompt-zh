// 悬停图片放大 / Image zoom on hover
// 分类包: lp (落地页与网页动效) · element 10 · 子分类: 悬停与点击交互
// 动画表现: 光标下的照片缓慢放大，外框尺寸不变，移开后缩回。
// 运动规格:
//   - 两个 222 × 118、圆角 8 的照片框；光标先停左侧再右侧，各约 1.5 秒。
//   - 悬停时内容在 0.5 秒内以三次缓出从中心缩放 1→1.15，移开后用 0.5 秒返回。
//   - 框外内容裁剪，对应 overflow: hidden；边框和说明文字不移动。
// 适用场景: 适合在作品、商品或文章图片网格中提示即将选中的项目。
// 标签: hover, zoom, image, scale, overflow-hidden, thumbnail, gallery
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/lp.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#082
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 10
// 10 ホバーで画像ズーム（枠はそのままで、中の写真だけがゆっくり拡大する）
CELLS[9] = (k, w, h, u) => {
  frame(k);
  const FX = [22, 267], FY = 16, FW = 222, FH = 118, ZOOM = 0.15;
  const cur = t => path([[0, 60, 160], [0.5, 120, 70], [1.8, 140, 84], [2.2, 370, 70], [3.4, 390, 84], [3.9, 470, 160], [4.8, 60, 160]], t);
  view(k, () => {
    for (let i = 0; i < 2; i++) {
      const inside = t => { const [x, y] = cur(t); return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH; };
      const q = trans(inside, u, 0.5);
      pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
      txt(k, ['夏の山小屋', '海辺のカフェ'][i], FX[i], FY + FH + 20, `800 13px ${JP}`, C.ink); bars(k, FX[i], FY + FH + 28, 150, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
