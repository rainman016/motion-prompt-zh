// 色差 / Chromatic aberration
// 分类包: photo (摄影光线与风格) · element 15 · 子分类: 镜头与相机效果
// 动画表现: 街景拆为红绿蓝三通道，以略不同尺寸重新叠加，人物、建筑和路灯边缘出现红青边，越靠两侧越明显。
// 运动规格:
//   - 场景 2：街角的建筑、树和路灯位于 y 132–162 的道路后方，路灯 x 114, 292, 494；人物站在左前方，脚底 x 176、y 200，高 124 px。
//   - 把图像分别乘以红 #FF0000、绿 #00FF00、蓝 #0000FF 得到三份，再在黑底上相加。
//   - 围绕中心缩放：红 1 + 2s、绿 1 + s、蓝 1，s = 0 → 0.02。
//   - 左右边缘红蓝最终相距约 11 px，中心偏移为 0。
//   - 效果在 0.4 秒至 1.4 秒进入，保持到 4.0 秒，在 4.8 秒前消失；每轮 5 秒，三次缓入缓出，约 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合旧镜头、VHS 或故障风格的不完美质感。
// 标签: chromatic aberration, color fringing, rgb split, lens aberration, fringe, glitch
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/photo.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#111
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 15
// 色収差: 赤・緑・青を少しずつ違う大きさで重ね、画面の端ほど縁に赤と青のずれが出る
CELLS[14] = cell((k, u) => {
  const a = env(u), s = 0.02 * a, cx = CW / 2, cy = CH / 2;
  if (s < 0.0005) { stageStreet(k, BASE); return; }
  const [S, ks] = layer('ca'); stageStreet(ks, BASE);
  k.fillStyle = '#000000'; k.fillRect(0, 0, CW, CH);
  for (const [col, sc] of [['#FF0000', 1 + 2 * s], ['#00FF00', 1 + s], ['#0000FF', 1]]) {
    const [L, kl] = layer('ca-' + col); kl.drawImage(S, 0, 0); kl.globalCompositeOperation = 'multiply'; kl.fillStyle = col; kl.fillRect(0, 0, CW, CH); kl.globalCompositeOperation = 'source-over';
    k.save(); k.globalCompositeOperation = 'lighter'; k.translate(cx, cy); k.scale(sc, sc); k.translate(-cx, -cy); k.drawImage(L, 0, 0); k.restore();
  }
});
