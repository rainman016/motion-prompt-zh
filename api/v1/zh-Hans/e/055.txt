// 气泡弹出框 / Popover
// 分类包: ui (UI 组件) · element 07 · 子分类: 弹出与覆盖
// 动画表现: 点击“详情”后右侧出现白色小面板，尖角指向按钮，点击内部“确定”关闭。
// 运动规格:
//   - 542×191 卡片，周期 4.5 秒。蓝色描边“详情”按钮为 108×36，位于 (24, 112)，光标于 1.0 秒点击。
//   - 显现量 a 在 1.0–1.28 秒由 0→1，使用 easeOutCubic；透明度 a，从右侧滑入 6 px。
//   - 面板为白色 300×120、圆角 10，位于 (156, 62)，边框 1 px，阴影模糊 22、y 偏移 6；左边缘 y 122–138 处有宽 10 的三角指向按钮。
//   - 内部有粗体标题、两行文字、70×26 描边按钮和 72×26 蓝色“确定”按钮。
//   - 关闭：光标在 3.0 秒前移至“确定”(402, 157)，3.3 秒点击；a 乘以 3.3–3.5 秒内的 (1 − easeInOutCubic)。
// 适用场景: 适合紧邻触发按钮的简短设置或确认操作。
// 标签: popover, popup, callout, bubble, flyout, overlay
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#055
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 07
// 07 ポップオーバー（Popover）
CELLS[6] = ui((w, h, t) => {
  pageLines(24, 16, 494);
  strokeRR(24, 112, 108, 36, 8, C.blue, 2); text('詳細', 78, 130, 15, C.blue, 600, 'center');
  fillRR(24, 160, 494, 22, 6, C.block);
  const a = eo(seg(t, 1.0, 1.28)) * (1 - eio(seg(t, 3.3, 3.5)));
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.translate(6 * (1 - a), 0);
    const px = 156, py = 62, pw = 300, ph = 120;
    shadowed(() => { fillRR(px, py, pw, ph, 10, '#fff'); }, 22, 6, .2); strokeRR(px, py, pw, ph, 10, C.border, 1);
    X.beginPath(); X.moveTo(px + 1, 122); X.lineTo(px - 10, 130); X.lineTo(px + 1, 138); X.fillStyle = '#fff'; X.fill();
    X.beginPath(); X.moveTo(px, 122); X.lineTo(px - 10, 130); X.lineTo(px, 138); X.strokeStyle = C.border; X.lineWidth = 1; X.stroke();
    bar(px + 18, py + 18, 130, C.ink, 11); bar(px + 18, py + 42, 250); bar(px + 18, py + 60, 200);
    strokeRR(px + 130, py + 82, 70, 26, 6, C.line2, 1.5); fillRR(px + 210, py + 82, 72, 26, 6, C.blue); text('OK', px + 246, py + 95, 14, '#fff', 700, 'center', NUMF);
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 84, 134], [1.6, 84, 134], [3.0, 402, 157], [3.5, 402, 157], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
