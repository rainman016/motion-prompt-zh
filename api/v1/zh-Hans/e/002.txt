// 交错入场 / Stagger
// 分类包: motion (动态图形基础) · element 02 · 子分类: 入场与文字
// 动画表现: 图表的柱条从左到右依次升起，每根略有延迟。
// 运动规格:
//   - 每 2.35 秒一轮。七根柱条，宽 44、间隔 20、高度为 52, 84, 68, 112, 92, 136, 158，依次循环使用蓝、珊瑚、黄、薄荷色。
//   - 柱条 i（从 0 计数）于 0.15 + 0.12·i 秒开始，使用 backOut（s = 2.6）在 0.5 秒内从 0 长到完整高度；底边固定，先超出再回落。
//   - 白点半径 6.5，位于柱顶上方 16 px；比对应柱条晚 0.34 秒，按弹簧公式 1 − e^(−7t)·cos(16t) 出现。
//   - 退场：柱条从 1.78 + 0.045·i 秒起，以三次缓入在 0.26 秒内缩回；白点先从 1.70 + 0.045·i 秒起，在 0.2 秒内消失。
// 适用场景: 适合列表行、图表柱条、卡片等成组元素，按顺序引导视线。
// 标签: stagger, sequence, delay, cascade, bar-chart, list, entrance
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/motion.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#002
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 02
// 02 スタガー（グラフの棒が、少しずつ時間をずらして順に立ち上がる）
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 棒の上の点（さらに遅れて出る）
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
