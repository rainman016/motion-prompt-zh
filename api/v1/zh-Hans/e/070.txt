// 加载旋转指示器 / Spinner
// 分类包: ui (UI 组件) · element 22 · 子分类: 导航与状态
// 动画表现: 点击“发送”后卡片蒙白，中央蓝色圆弧旋转，完成后变为对勾并淡出。
// 运动规格:
//   - 542×191 卡片，周期 4.5 秒。光标于 1.0 秒点击底部居中的 120×40“发送”按钮。
//   - 1.05–1.2 秒线性淡入透明度 0.78 的白色全卡片遮罩和旋转指示器。
//   - 指示器中心 (271, 92)，浅色环半径 26、线宽 6；上层蓝色圆弧长 243°，即 1.35π，圆形端点，每秒转 1.3 圈，即 468°/s。
//   - 2.95–3.05 秒旋转环淡出；2.95–3.2 秒带白色对勾的半径 28 蓝圆由 0.7 → 1 放大，使用 easeOutCubic。
//   - 3.75–4.0 秒全部线性淡出。
// 适用场景: 适合发送、保存等时长未知的短暂等待。
// 标签: spinner, loading, loader, activity-indicator, progress-circle, busy
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://rainman016.github.io/motion-prompt-zh/api/v1/kit/ui.js
// 动画演示: https://rainman016.github.io/motion-prompt-zh/?lang=zh-Hans#070
//
// cells[i](k, w, h, u)：k 为 CanvasRenderingContext2D，调用方已清空并包裹 save/restore；w、h 为尺寸，u 为 [0, cycle) 内的秒数。画面仅由 u 决定。
// 以下为元素代码，CELLS、常量和辅助函数位于共享 kit 中。

// @element 22
// 22 スピナー（Spinner）
CELLS[21] = ui((w, h, t) => {
  const load = seg(t, 1.05, 1.2) * (1 - seg(t, 2.95, 3.05));
  const done = seg(t, 2.95, 3.15) * (1 - seg(t, 3.75, 4.0));
  pageLines(24, 18, 494); strokeRR(24, 90, 494, 34, 7, C.line2, 1.5);
  fillRR(w / 2 - 60, 140, 120, 40, 8, C.blue); text('送信', w / 2, 160, 16, '#fff', 600, 'center');
  const ov = Math.max(load, done);
  if (ov > 0) {
    X.fillStyle = `rgba(255,255,255,${.78 * ov})`; X.fillRect(0, 0, w, h);
    if (load > 0) { X.save(); X.globalAlpha = load; ring(w / 2, 92, 26, '#e3e8f5', 6); X.beginPath(); const a0 = t * Math.PI * 2 * 1.3; X.arc(w / 2, 92, 26, a0, a0 + Math.PI * 1.35); X.strokeStyle = C.blue; X.lineWidth = 6; X.lineCap = 'round'; X.stroke(); X.restore(); }
    if (done > 0) { X.save(); X.globalAlpha = done; const s = .7 + .3 * eo(seg(t, 2.95, 3.2)); X.translate(w / 2, 92); X.scale(s, s); circle(0, 0, 28, C.blue); check(0, 0, 11, '#fff', 4); X.restore(); }
  }
  pointer([[.3, w + 30, h + 30], [.95, w / 2 + 6, 166], [1.5, w / 2 + 6, 166], [2.2, w + 30, h + 30]], [1.0], t);
});
